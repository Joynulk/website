# HOW TO USE THIS FILE
# The automatic tools cannot create files under .github/workflows (GitHub blocks it), so add it by hand once, on the main branch:
#   GitHub repo (branch: main) -> Add file -> Create new file -> name it   .github/workflows/pages.yml
#   -> paste everything BELOW this comment block -> Commit.
# Then: Settings -> Pages -> Source = "GitHub Actions", and run the "Deploy site" workflow from the Actions tab.
#
# How it works: the static pages come from the main branch, the React app comes from the react-app branch.
# After new work lands on the react-app branch, run the workflow again (Actions -> Deploy site -> Run workflow).
#
name: Deploy site

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      # static pages (main branch)
      - uses: actions/checkout@v4

      # React app source (react-app branch)
      - uses: actions/checkout@v4
        with:
          ref: react-app
          path: _react

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Build the React app
        working-directory: _react/react-app
        run: |
          npm install --no-audit --no-fund
          npm run build

      - name: Put the site together
        run: |
          mkdir _site
          rsync -a --exclude='.git' --exclude='.github' --exclude='react-app' --exclude='_react' --exclude='Doc' --exclude='supabase' --exclude='_site' ./ _site/
          mkdir -p _site/app
          cp -r _react/react-app/dist/. _site/app/

      - uses: actions/configure-pages@v5

      - uses: actions/upload-pages-artifact@v3
        with:
          path: _site

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
